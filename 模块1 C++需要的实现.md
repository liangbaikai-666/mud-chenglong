## 模块1 C++ 类设计

基于修补版md和 `game_api.h`，以下是模块1（角色属性与战斗核心引擎）的C++类设计。**返回值与实现思路写在注释中。**

cpp

```cpp
#ifndef MODULE1_CHARACTER_BATTLE_H
#define MODULE1_CHARACTER_BATTLE_H

#include "game_api.h"  // 包含共享枚举、前置声明、其他模块接口声明
#include <string>
#include <vector>

// ==================== Player 类 ====================
class Player {
private:
    // ---- 属性成员（来自功能1） ----
    int current_hp;
    int max_hp;
    int current_mp;
    int level;
    int exp;
    int base_strength;
    float base_hit_rate;
    // 注：money 已移至模块4，此处不再保留

public:
    // ---- 构造函数 ----
    Player() : current_hp(100), max_hp(100), current_mp(100), level(1), exp(0),
               base_strength(15), base_hit_rate(0.95f) {}

    // ---- 功能1：属性获取与修改 ----
    /**
     * 获取指定属性值
     * @param attr_name 属性名常量（来自 shared_attributes.h，如 Attr::HP）
     * @return int 对应属性的当前值
     * 实现思路：if-else 或 map 查找，返回对应成员变量的值
     */
    int get_attr(const std::string& attr_name) const;

    /**
     * 修改属性（增加或设置）
     * @param attr_name 属性名常量
     * @param value 变化量或目标值
     * @param mode "add"=增减（受范围限制），"set"=直接设置（受上下限限制）
     * 实现思路：根据attr_name找到对应成员，按mode修改，并clamp到合法范围
     *          例如 strength 范围1~99，hp 范围0~max_hp
     */
    void modify_attr(const std::string& attr_name, int value, const std::string& mode = "add");

    // ---- 功能2：HP/MP实时计算与扣减 ----
    /**
     * 承受伤害
     * @param amount 伤害数值
     * @return bool true=存活，false=死亡（current_hp降为0或以下）
     * 实现思路：current_hp = max(0, current_hp - amount);
     *          若 current_hp == 0，设置 is_dead = true，触发 check_death()
     */
    bool take_damage(int amount);

    /**
     * 恢复HP
     * @param amount 恢复量
     * @return void
     * 实现思路：current_hp = min(max_hp, current_hp + amount);
     */
    void heal(int amount);

    /**
     * 消耗MP
     * @param amount 消耗量
     * @return bool true=足够并扣除，false=不足
     * 实现思路：if current_mp >= amount, 扣除并返回true; else 返回false
     */
    bool consume_mp(int amount);

    /**
     * 恢复MP
     * @param amount 恢复量
     * @return void
     * 实现思路：current_mp = min(500, current_mp + amount);
     */
    void restore_mp(int amount);

    // ---- 功能1扩展：经验与升级 ----
    /**
     * 获得经验值
     * @param amount 获得的经验
     * @return void
     * 实现思路：exp += amount; 循环检查是否达到 level * 100 阶梯，
     *          若达到则 level++，exp -= 所需经验，提升 max_hp 等属性
     */
    void gain_exp(int amount);

    // getter（供战斗引擎等外部读取）
    int get_current_hp() const { return current_hp; }
    int get_max_hp() const { return max_hp; }
    int get_level() const { return level; }
    int get_strength() const { return base_strength; }
};

// ==================== BattleEngine 类 ====================
class BattleEngine {
private:
    // ---- 功能3成员 ----
    GameState battle_state;          // NOT_IN_BATTLE, PLAYER_TURN, ENEMY_TURN, BATTLE_END
    int turn_count;
    std::vector<std::string> action_queue;  // 动作队列
    std::vector<Enemy*> current_enemies;
    bool player_action_done;
    int action_points;               // 本回合剩余行动点

    // ---- 功能4成员 ----
    std::map<int, int> cooldown_dict;     // 技能/符咒冷却
    int global_turn_counter;

    Player* player;                  // 玩家指针

public:
    BattleEngine(Player* p) : player(p), battle_state(GameState::PLAYING),
                              turn_count(0), player_action_done(false),
                              action_points(1), global_turn_counter(0) {}

    // ---- 功能3：战斗流程控制 ----
    /**
     * 开始战斗
     * @param enemy_list 敌人列表
     * @return void
     * 实现思路：设置 battle_state = PLAYER_TURN，清空队列，turn_count=1，
     *          根据玩家手持兔符咒设置 action_points（有兔符咒则为2）
     */
    void start_battle(const std::vector<Enemy*>& enemy_list);

    /**
     * 处理玩家动作
     * @param action_type 动作类型（攻击/技能/道具）
     * @param target 目标指针
     * @return void
     * 实现思路：将动作加入 action_queue，消耗1点 action_points；
     *          若 action_points 耗尽，标记 player_action_done = true 并调用 end_turn()
     */
    void player_action(const std::string& action_type, Entity* target);

    /**
     * 敌人回合行动
     * @return void
     * 实现思路：遍历 current_enemies，根据AI决定动作加入队列
     */
    void enemy_action();

    /**
     * 依次执行动作队列
     * @return void
     * 实现思路：遍历 action_queue，计算伤害/效果。
     *          【跨模块调用】若动作涉及符咒，调用模块2：
     *          int dmg = TalismanManager::trigger_talisman(talisman_id, caster);
     *          然后调用 player->take_damage(dmg) 或 enemy->take_damage(dmg)
     */
    void process_actions();

    /**
     * 结束当前回合
     * @return void
     * 实现思路：调用 tick_cooldowns()，减少Buff/Debuff持续回合，切换 battle_state
     */
    void end_turn();

    /**
     * 检查战斗是否结束
     * @return void
     * 实现思路：若所有敌人HP<=0，设置 battle_state = BATTLE_END，触发胜利回调；
     *          若玩家 is_dead，触发失败回调
     */
    void check_battle_end();

    // ---- 功能4：冷却计数 ----
    /**
     * 开始冷却
     * @param key 技能/符咒ID
     * @param turns 冷却回合数
     * @return void
     * 实现思路：cooldown_dict[key] = turns;
     */
    void start_cooldown(int key, int turns);

    /**
     * 每玩家回合结束时减少所有冷却
     * @return void
     * 实现思路：遍历 cooldown_dict，所有 value > 0 的减1
     */
    void tick_cooldowns();

    // ---- 功能6：外部接口（供其他模块调用） ----
    /**
     * 兔符咒额外行动点
     * @return void
     * 实现思路：【限制】仅在玩家回合（battle_state == PLAYER_TURN）且回合未结束时，
     *          增加 action_points += 1
     */
    void grant_extra_action();

    /**
     * 获取当前行动点数
     * @return int 当前 action_points 值
     */
    int get_action_points() const { return action_points; }
};

#endif // MODULE1_CHARACTER_BATTLE_H
```
