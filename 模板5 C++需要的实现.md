```cpp
#ifndef MODULE5_LEVEL_FLOW_H
#define MODULE5_LEVEL_FLOW_H

#include "game_api.h"  // 包含共享枚举、前置声明、结构体定义（LevelConfig, DialogueNode, RewardConfig等）
#include <string>
#include <vector>
#include <set>
#include <map>

// 前置声明其他模块的管理器（避免循环依赖，具体定义在各自的头文件中）
class TalismanManager;
class StatusEffectManager;
class InventoryManager;
class BattleEngine;

// ==================== 关卡管理器（功能1、2、5、6部分） ====================
class LevelManager {
private:
    // ---- 功能1：进度数据 ----
    int current_layer;                              // 当前层（1~4）
    std::set<std::string> cleared_levels;           // 已通关的关卡ID集合
    std::set<std::string> unlocked_levels;          // 已解锁可进入的关卡ID集合
    bool all_talismans_collected;                   // 是否集齐全套十二符咒

    // ---- 关卡配置表 ----
    // LevelConfig 定义在 game_api.h 中，含 id, layer, prerequisites, waves, difficulty_coef, rewards
    std::map<std::string, LevelConfig> level_configs;

    // ---- 功能2：分支选择 ----
    std::string selected_branch;                    // 当前选择的分支ID
    int enemy_wave_index;                           // 当前波次索引

    // ---- 功能5：结算 ----
    BattleResult battle_result;                     // 战斗结果枚举（WIN/LOSE/FLEE）
    RewardConfig reward_pool;                       // 奖励配置（定义在 game_api.h）
    bool level_clear_flag;                          // 是否首次通关该关卡

    // ---- 功能6：跨模块引用（由外部注入） ----
    TalismanManager* talisman_mgr;                  // 模块2符咒管理器指针
    InventoryManager* inventory_mgr;                // 模块4背包管理器指针
    BattleEngine* battle_engine;                    // 模块1战斗引擎指针

public:
    /**
     * 构造函数
     * 实现思路：初始化所有进度为初始状态，指针置空
     */
    LevelManager()
        : current_layer(1), all_talismans_collected(false),
          selected_branch(""), enemy_wave_index(0),
          battle_result(BattleResult::LOSE), level_clear_flag(false),
          talisman_mgr(nullptr), inventory_mgr(nullptr), battle_engine(nullptr) {
        load_level_configs();
    }

    // ---- 注入跨模块管理器 ----
    void set_talisman_manager(TalismanManager* mgr) { talisman_mgr = mgr; }
    void set_inventory_manager(InventoryManager* mgr) { inventory_mgr = mgr; }
    void set_battle_engine(BattleEngine* engine) { battle_engine = engine; }

    // ---- 功能1：关卡树与进度管理 ----
    /**
     * 初始化进度
     * @return void
     * 实现思路：清空 cleared_levels 和 unlocked_levels，将L1的3个关卡ID加入 unlocked_levels，
     *          设置 current_layer = 1。
     */
    void init_progress();

    /**
     * 检查关卡是否已通关
     * @param level_id 关卡ID
     * @return bool true=已通关，false=未通关
     * 实现思路：返回 cleared_levels.count(level_id) > 0
     */
    bool is_level_cleared(const std::string& level_id) const;

    /**
     * 检查是否可以进入关卡
     * @param level_id 关卡ID
     * @return bool true=可以进入，false=条件不满足
     * 实现思路：根据关卡所属层检查解锁条件：
     *          - L2关卡：检查L1的3个关卡是否全部在 cleared_levels 中
     *          - L3关卡：统计L2中已通关数量，需≥2
     *          - L4关卡：统计L3中已通关数量，需≥1
     *          同时检查 level_id 是否在 unlocked_levels 中。
     */
    bool can_enter_level(const std::string& level_id) const;

    /**
     * 关卡通关回调
     * @param level_id 通关的关卡ID
     * @return void
     * 实现思路：将 level_id 加入 cleared_levels。然后按规则检查解锁：
     *          - 若L1全部打通（cleared_levels包含L1所有3关），将L2的4个关卡加入 unlocked_levels
     *          - 若L2通关数≥2，将L3的5个关卡加入 unlocked_levels
     *          - 若L3通关数≥1，将L4的1个关卡加入 unlocked_levels
     *          更新 current_layer 为下一层（若适用）。
     */
    void on_level_cleared(const std::string& level_id);

    /**
     * 自动补全缺失符咒
     * @return void
     * 实现思路：进入终战前调用。遍历1~12所有符咒ID，若玩家背包中没有，
     *          直接调用 talisman_mgr->add_to_backpack(id) 加入。
     */
    void auto_grant_missing_talismans();

    // ---- 功能2：关卡进入与分支 ----
    /**
     * 加载关卡配置
     * @param level_id 关卡ID
     * @return LevelConfig 关卡配置结构体
     * 实现思路：返回 level_configs[level_id]
     */
    LevelConfig load_level_config(const std::string& level_id);

    /**
     * 选择分支
     * @param branch_id 分支ID
     * @return void
     * 实现思路：调用 can_enter_level 校验，若通过则设置 selected_branch = branch_id
     */
    void select_branch(const std::string& branch_id);

    /**
     * 生成敌人波次
     * @param wave_index 波次索引
     * @return std::vector<Enemy> 敌人列表
     * 实现思路：根据当前关卡配置和 difficulty_coef，生成对应波次的敌人。
     *          实际战斗由 battle_engine->start_battle 接管。
     */
    std::vector<Enemy> generate_enemy_wave(int wave_index);

    /**
     * 获取当前层所有可选分支
     * @return std::vector<std::string> 分支ID列表
     * 实现思路：遍历 level_configs，收集与 current_layer 匹配且已解锁的关卡ID
     */
    std::vector<std::string> get_available_branches() const;

    // ---- 功能5：战斗结算 ----
    /**
     * 战斗结束处理
     * @param result 战斗结果
     * @param defeated 被击败的敌人列表
     * @return void
     * 实现思路：若 result == WIN，调用 calculate_rewards 计算奖励，
     *          调用 inventory_mgr->add_money 和 inventory_mgr->add_item 发放。
     *          若 level_clear_flag 为true，触发关卡结算对话（调用DialogueManager）。
     *          调用 on_level_cleared 更新进度。
     */
    void on_battle_end(BattleResult result, const std::vector<Enemy*>& defeated);

    /**
     * 计算奖励
     * @param enemy_difficulty 敌人难度系数
     * @return RewardConfig 奖励配置
     * 实现思路：根据 enemy_difficulty 和 reward_pool 基数计算最终金钱和掉落
     */
    RewardConfig calculate_rewards(int enemy_difficulty);

    // ---- 功能6：与战斗引擎交互 ----
    /**
     * 为关卡启动战斗
     * @param level_id 关卡ID
     * @return void
     * 实现思路：加载关卡配置，生成第一波敌人，调用 battle_engine->start_battle(enemy_list)
     */
    void start_battle_for_level(const std::string& level_id);

    /**
     * 接收战斗结果回调
     * @param result 战斗结果
     * @return void
     * 实现思路：保存 result，调用 on_battle_end
     */
    void on_battle_result_received(BattleResult result);

    /**
     * 剧情强制添加状态
     * @param status_id 状态ID
     * @return void
     * 实现思路：调用 status_mgr->add_status(player, status_id, duration)
     *          （status_mgr为外部注入的StatusEffectManager指针）
     */
    void apply_story_status(const std::string& status_id);

    /**
     * 发放道具奖励
     * @param item_id 道具ID
     * @param count 数量
     * @return void
     * 实现思路：调用 inventory_mgr->add_item(item_id, count)
     */
    void grant_item_reward(const std::string& item_id, int count);

private:
    // ---- 内部辅助方法 ----
    /**
     * 加载关卡配置表
     * 实现思路：从配置文件或硬编码读取所有关卡数据到 level_configs
     */
    void load_level_configs();
};

// ==================== 对话管理器（功能3） ====================
class DialogueManager {
private:
    // 键为NPC名+进度阶段，值为对话节点列表
    // DialogueNode 定义在 game_api.h 中
    std::map<std::string, std::vector<DialogueNode>> npc_dialogue_table;
    std::string current_stage;                      // 当前剧情阶段标识
    std::string active_npc;                         // 当前交互的NPC ID

public:
    /**
     * 构造函数
     * 实现思路：初始化 current_stage 和 active_npc 为空，加载对话表
     */
    DialogueManager() : current_stage(""), active_npc("") {
        load_dialogue_table();
    }

    // ---- 功能3：NPC对话与剧情 ----
    /**
     * 获取NPC对话
     * @param npc_id NPC ID
     * @return std::vector<DialogueNode> 对话节点列表
     * 实现思路：根据 current_stage 和 npc_id 拼接键，从 npc_dialogue_table 中查找返回
     */
    std::vector<DialogueNode> get_npc_dialogue(const std::string& npc_id) const;

    /**
     * 推进剧情阶段
     * @param new_stage 新阶段标识
     * @return void
     * 实现思路：更新 current_stage = new_stage，触发新对话分支
     */
    void advance_stage(const std::string& new_stage);

    /**
     * NPC交互
     * @param npc_id NPC ID
     * @return std::string 对话内容文本
     * 实现思路：设置 active_npc = npc_id，返回 get_npc_dialogue(npc_id) 的首条文本
     */
    std::string on_npc_interact(const std::string& npc_id);

    /**
     * 触发引导提示
     * @param type 引导类型（GuideType枚举，定义在 game_api.h）
     * @return std::string 提示文本
     * 实现思路：根据 type 返回对应的小玉引导文本
     */
    std::string trigger_guide_hint(GuideType type) const;

private:
    void load_dialogue_table();                     // 加载对话配置
};

// ==================== 全局游戏状态管理器（功能4） ====================
class GameStateManager {
private:
    GameState game_state;                           // 枚举：PLAYING, STORY_DEATH, VICTORY, GAME_OVER

public:
    /**
     * 构造函数
     * 实现思路：初始化 game_state = PLAYING
     */
    GameStateManager() : game_state(GameState::PLAYING) {}

    // ---- 功能4：状态管理 ----
    /**
     * 获取当前游戏状态
     * @return GameState 当前状态枚举
     * 实现思路：返回 game_state
     */
    GameState get_game_state() const;

    /**
     * 检查即死条件
     * @return void
     * 实现思路：检查特定剧情节点是否缺少关键道具/符咒，若满足则设置 game_state = STORY_DEATH
     */
    void check_story_death();

    /**
     * 通关胜利
     * @return void
     * 实现思路：设置 game_state = VICTORY
     */
    void on_victory();

    /**
     * 游戏结束
     * @return void
     * 实现思路：设置 game_state = GAME_OVER
     */
    void on_game_over();
};

#endif // MODULE5_LEVEL_FLOW_H
```
