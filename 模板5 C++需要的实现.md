```cpp
#ifndef MODULE5_IMPL_H
#define MODULE5_IMPL_H

#include "game_api.h"       // 包含 ILevelManager、IDialogueManager、IGameStateManager 等

#include <vector>
#include <map>
#include <string>

// ==================== 关卡节点结构体（模块5内部使用） ====================
struct LevelNode {
    int layer;                      // 关卡层数（1~4）
    std::string level_id;           // 关卡唯一标识
    bool unlocked;                  // 是否已解锁
    bool cleared;                   // 是否已通关
    std::vector<std::string> enemy_groups;  // 敌人组配置
    RewardConfig reward;            // 通关奖励
};

// ==================== LevelManager 类（继承 ILevelManager） ====================
class LevelManager : public ILevelManager {
private:
    // ---- 核心数据 ----
    std::map<std::string, LevelNode> level_map;   // 关卡表
    std::string current_level_id;                 // 当前关卡ID
    int current_layer;                            // 当前层数

    // ---- 跨模块指针 ----
    IBattleEngine* battle_mgr;
    ITalismanManager* talisman_mgr;
    IStatusEffectManager* status_mgr;
    IInventoryManager* inventory_mgr;

public:
    LevelManager();

    // ==================== ILevelManager 接口实现 ====================
    /**
     * 初始化关卡进度
     * @return void
     * 实现思路：清空 level_map，创建 L1~L4 关卡节点，设置初始状态（L1解锁，其余锁定）
     */
    void init_progress() override;

    /**
     * 查询是否可以进入指定关卡
     * @param level_id 关卡ID
     * @return bool true=可进入，false=未解锁或条件不足
     * 实现思路：查找 level_map[level_id]，检查 unlocked 标记及前置条件
     */
    bool can_enter_level(const std::string& level_id) const override;

    /**
     * 关卡通关回调
     * @param level_id 关卡ID
     * @return void
     * 实现思路：设置 level_map[level_id].cleared = true，
     *          根据解锁规则解锁后续关卡，调用 inventory_mgr 发放奖励
     */
    void on_level_cleared(const std::string& level_id) override;

    /**
     * 自动补全缺失符咒（由关卡/剧情触发）
     * @return void
     * 实现思路：遍历全部12个 TalismanID，若 talisman_mgr->has_talisman(id) 为 false，
     *          则调用 talisman_mgr->add_to_backpack(id) 补全
     */
    void auto_grant_missing_talismans() override;

    /**
     * 为当前关卡启动战斗
     * @return void
     * 实现思路：根据 current_level_id 查找敌人配置，生成敌人列表，
     *          调用 battle_mgr->start_battle(enemies)
     */
    void start_battle_for_level() override;

    /**
     * 战斗结束回调
     * @param result 战斗结果（BattleResult）
     * @param enemies 参与战斗的敌人列表
     * @return void
     * 实现思路：若 result == WIN，调用 on_level_cleared(current_level_id)；
     *          若 result == LOSE，通知 GameStateManager 处理失败逻辑
     */
    void on_battle_end(BattleResult result, const std::vector<Enemy*>& enemies) override;

    /**
     * 获取当前关卡层数
     * @return int 当前层数（1~4）
     * 实现思路：return current_layer;
     */
    int get_current_layer() const override;

    /**
     * 查询是否所有符咒已收集
     * @return bool true=全部收集，false=仍有缺失
     * 实现思路：遍历12个 TalismanID，检查 talisman_mgr->has_talisman(id)
     */
    bool is_all_talismans_collected() const override;

    // ==================== 跨模块指针注入 ====================
    void set_battle_engine(IBattleEngine* mgr);
    void set_talisman_manager(ITalismanManager* mgr);
    void set_status_effect_manager(IStatusEffectManager* mgr);
    void set_inventory_manager(IInventoryManager* mgr);
};

// ==================== DialogueManager 类（继承 IDialogueManager） ====================
class DialogueManager : public IDialogueManager {
private:
    // ---- 核心数据 ----
    std::map<std::string, DialogueNode> dialogue_map;   // NPC对话表
    std::string current_npc_id;                          // 当前交互NPC
    int current_stage;                                   // 当前剧情阶段

    // ---- 跨模块指针 ----
    IGameStateManager* game_state_mgr;

public:
    DialogueManager();

    // ==================== IDialogueManager 接口实现 ====================
    /**
     * 获取NPC对话内容
     * @param npc_id NPC唯一标识
     * @return std::string 对话文本
     * 实现思路：查找 dialogue_map[npc_id]，根据 current_stage 返回对应文本
     */
    std::string get_npc_dialogue(const std::string& npc_id) const override;

    /**
     * 推进剧情阶段
     * @return void
     * 实现思路：current_stage++，触发阶段变化相关的剧情事件
     */
    void advance_stage() override;

    /**
     * NPC交互回调
     * @param npc_id NPC唯一标识
     * @return void
     * 实现思路：设置 current_npc_id = npc_id，调用 get_npc_dialogue 获取文本并输出
     */
    void on_npc_interact(const std::string& npc_id) override;

    /**
     * 触发引导提示
     * @param type 引导类型（GuideType 枚举）
     * @return void
     * 实现思路：根据 type 从配置中获取对应提示文本并输出到UI/日志
     */
    void trigger_guide_hint(GuideType type) override;

    // ==================== 跨模块指针注入 ====================
    void set_game_state_manager(IGameStateManager* mgr);
};

// ==================== GameStateManager 类（继承 IGameStateManager） ====================
class GameStateManager : public IGameStateManager {
private:
    // ---- 核心数据 ----
    GameState current_state;                  // 当前全局游戏状态

    // ---- 跨模块指针 ----
    IBattleEngine* battle_mgr;
    ILevelManager* level_mgr;

public:
    GameStateManager();

    // ==================== IGameStateManager 接口实现 ====================
    /**
     * 获取当前游戏状态
     * @return GameState 枚举值
     * 实现思路：return current_state;
     */
    GameState get_game_state() const override;

    /**
     * 检查剧情死亡条件
     * @return bool true=触发剧情死亡，false=未触发
     * 实现思路：检查玩家HP是否为0且处于特定剧情节点，若是则设置 current_state = STORY_DEATH
     */
    bool check_story_death() override;

    /**
     * 胜利处理
     * @return void
     * 实现思路：设置 current_state = VICTORY，触发通关流程
     */
    void on_victory() override;

    /**
     * 游戏结束处理
     * @return void
     * 实现思路：设置 current_state = GAME_OVER，清理资源，触发结束画面
     */
    void on_game_over() override;

    // ==================== 跨模块指针注入 ====================
    void set_battle_engine(IBattleEngine* mgr);
    void set_level_manager(ILevelManager* mgr);
};

#endif // MODULE5_IMPL_H
```

------

**说明：**

- 头文件引用仅 `#include "game_api.h"`，不引用其他头文件
- 以上仅为 `.h`头文件框架，所有方法只声明、注释返回值意义与实现思路，不含 .cpp 实现代码
- 组员根据注释中的"实现思路"自行编写 `.cpp`文件
- 跨模块调用全部通过抽象基类指针
- 符咒ID、状态ID、道具ID、引导类型等全部使用 `game_api.h`中定义的 `enum class`类型
- 关卡奖励发放通过 `IInventoryManager::add_item(ItemID, int)`/ `add_money(int)`
- 符咒补全通过 `ITalismanManager::add_to_backpack(TalismanID)`
